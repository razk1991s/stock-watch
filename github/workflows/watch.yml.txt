name: Stock Watch
on:
  schedule:
    - cron: "25 14 * * 1-5"   # 16:25 Israel
    - cron: "55 17 * * 1-5"   # 19:55 Israel
  workflow_dispatch:
jobs:
  watch:
    runs-on: ubuntu-latest
    timeout-minutes: 240
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install requests
      - run: python stock_watch.py
        env:
          FINNHUB_KEY: ${{ secrets.FINNHUB_KEY }}
          NTFY_TOPIC: ${{ secrets.NTFY_TOPIC }}
          RUN_MINUTES: "210"
